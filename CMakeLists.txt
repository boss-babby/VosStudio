cmake_minimum_required(VERSION 3.16)
project(VOSStudioNative CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
file(GLOB CORE src/core/*.cpp)
file(GLOB WIN src/win/*.cpp)
add_executable(VOSStudio WIN32 ${CORE} ${WIN} res/vosstudio.rc)
target_compile_definitions(VOSStudio PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(VOSStudio PRIVATE d3d11 d2d1 dwrite d3dcompiler dxgi windowscodecs ole32 oleaut32 uuid shell32 comdlg32 winhttp dbghelp dwmapi crypt32 user32 gdi32)
if(MINGW)
  target_link_options(VOSStudio PRIVATE -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
  target_compile_options(VOSStudio PRIVATE /utf-8 /W3 /permissive- /EHsc)
endif()
